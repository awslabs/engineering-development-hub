# Let sackd read the Slurm key of AWS PCS cluster {{ pcs_cluster_name }} ({{ cluster_id }}).

#   sudo bash pcs-key-permissions-{{ cluster_id }}.sh

KEY_DIR="{{ _key_dir }}"

fail() { echo "ERROR: $*" >&2; exit 1; }

#!/bin/bash
# Register AWS PCS cluster {{ pcs_cluster_name }} ({{ cluster_id }}) as an EDH scheduler.

{{ _sackd_setup_script }}echo "==> Step 4 - confirm connectivity to the SLURMCTLD"
