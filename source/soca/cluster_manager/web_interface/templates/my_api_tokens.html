{{ _("Query a job") }}
{{ _("Using Bash") }}

curl -k -X GET \
   -H "X-EDH-TOKEN: $YOUR_SCOPED_TOKEN" \
   -H "X-EDH-USER: {{ user }}" \
   {{ scheduler_host }}api/scheduler/job?job_id=%YOUR_JOB_ID%&scheduler_id={{openpbs_scheduler}}
{{ _("Using Python") }}

import requests
hdrs = {"X-EDH-TOKEN": "YOUR_SCOPED_TOKEN",
        "X-EDH-USER": "{{ user }}"}
api_endpoint = "{{ scheduler_host }}api/scheduler/job"
data = {"job_id": "%YOUR_JOB_ID%", "scheduler_id": "{{openpbs_scheduler}}"}
req = requests.get(api_endpoint, headers=hdrs, params=data, verify=False)
print(req.status_code)
print(req.json())
